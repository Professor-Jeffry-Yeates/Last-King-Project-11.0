def myFunction():
    name = "Corn.chips"
    folder = "Problems"
    cwd = os.getcwd()
    path = pathlib.Path( cwd, folder, name )
    with open( path, "a") as writer:
        writer.write("yeah\n")
    with open( path, "r") as reader:
